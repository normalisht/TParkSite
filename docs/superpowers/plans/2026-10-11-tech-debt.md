class PermissionsPolicyMiddleware:
    """Запрещает странице и встроенным фреймам камеру, микрофон, платежи и USB — сайту они не нужны.
    Геолокацию не трогаем: «Моё местоположение» в карте Яндекса."""

```python
MAX_PAGE_BYTES = 5 * 1024 * 1024  # страница отзывов — сотни килобайт; больше — не то, что ждём


def is_yandex_url(url: str) -> bool:
    parts = urlsplit(url)
    host = parts.hostname or ""
    return parts.scheme == "https" and (host == "yandex.ru" or host.endswith(".yandex.ru"))


class YandexOnlyRedirects(HTTPRedirectHandler):
    """Редиректы (капча, смена адреса) — только внутри yandex.ru: запрос идёт с сервера по ссылке из админки."""

    def redirect_request(self, req, fp, code, msg, headers, newurl):
        if not is_yandex_url(newurl):
            raise ReviewFetchError("Яндекс перенаправил на другой сайт — заполните отзыв вручную.")
        return super().redirect_request(req, fp, code, msg, headers, newurl)


_opener = build_opener(YandexOnlyRedirects)


def read_limited(response, limit: int = MAX_PAGE_BYTES) -> bytes:
    data = response.read(limit + 1)
    if len(data) > limit:
        raise ReviewFetchError("Страница Яндекса слишком большая — заполните отзыв вручную.")
    return data
```

```python
SCHEMA_DAYS = {
    "Mo": "Monday",
    "Tu": "Tuesday",
    "We": "Wednesday",
    "Th": "Thursday",
    "Fr": "Friday",
    "Sa": "Saturday",
    "Su": "Sunday",
}
_ORDER = list(DAYS)


def _expand(spec: str) -> list[str]:
    """«Mo-Fr,Su» → [Mo, Tu, We, Th, Fr, Su]; диапазон через воскресенье («Fr-Mo») тоже разворачивается."""
    days = []
    for part in spec.split(","):
        first, _, last = part.partition("-")
        start, end = _ORDER.index(first), _ORDER.index(last or first)
        days += _ORDER[start : end + 1] if start <= end else _ORDER[start:] + _ORDER[: end + 1]
    return days


def opening_hours_spec(value: str) -> list[dict]:
    """`OpeningHoursSpecification` для JSON-LD; если хоть одно правило не разобрать — пустой список."""
    specs = []
    for rule in filter(None, (r.strip() for r in (value or "").split(";"))):
        match = _RULE.match(rule)
        if not match:
            return []
        specs.append(
            {
                "@type": "OpeningHoursSpecification",
                "dayOfWeek": [f"https://schema.org/{SCHEMA_DAYS[d]}" for d in _expand(match["days"])],
                "opens": match["start"],
                "closes": match["end"],
            }
        )
    return specs
```
